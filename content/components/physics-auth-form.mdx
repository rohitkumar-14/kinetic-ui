## Usage

This form uses \`matter-js\` to create a physical environment that reacts to user input. It is highly engaging and gamifies the process of choosing a strong password.

```tsx
import { PhysicsAuthForm } from "@/components/creative/physics-auth-form";

export function AuthPage() {
  const handleLogin = (email: string, pass: string) => {
    console.log("Authenticating...", email);
  };

  return (
    <div className="w-full flex justify-center p-8">
      <PhysicsAuthForm onLogin={handleLogin} />
    </div>
  );
}
```
